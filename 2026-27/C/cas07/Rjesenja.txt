SELECT K1.ID, K2.ID FROM Klijent K1, Klijent K2 WHERE K1.ID < K2.ID 
AND EXISTS(
	SELECT * FROM Iznajmljivanje I1, Iznajmljivanje I2 WHERE 
	I1.Klijent_ID = K1.ID AND
	I2.Klijent_ID = K2.ID AND
	I1.Knjiga_ID = I2.Knjiga_ID
)

SELECT K1.ID, K2.ID FROM Klijent K1, Klijent K2 WHERE K1.ID < K2.ID 
AND EXISTS(
	SELECT K.ID FROM Knjiga K WHERE 
	EXISTS(
		SELECT * FROM Iznajmljivanje WHERE Klijent_ID = K1 AND 
		Knjiga_ID = K.ID
	) AND 
	EXISTS(
		SELECT * FROM Iznajmljivanje WHERE Klijent_ID = K2 AND 
		Knjiga_ID = K.ID
	)
)
